/* ---------------- 导航：左侧栏 / 移动端抽屉 / URL hash 同步 ---------------- */ var TABS = ['overview','sensors','services','processes','files','terminal']; var TITLES = {overview:'概览', sensors:'传感器', services:'服务', processes:'进程', files:'文件', terminal:'终端'}; var current = 'overview'; var appEl = $('app'); function switchTab(name, silent){ if(TABS.indexOf(name) < 0) name = 'overview'; /* 离开传感器页就把持续采集停掉：采样会真唤醒硬件，不能让它留在后台烧电 */ if(name !== 'sensors' && typeof stopSensorLive === 'function') stopSensorLive(); /* 离开文件页要收掉预览弹窗：弹窗是 fixed 定位，不收拾就会浮在新页面上面； 更要紧的是视频/音频会继续出声。_js_files 在本文件之后加载，所以带 typeof 判空。 */ if(name !== 'files' && typeof closeViewer === 'function' && $('viewMask').classList.contains('show')) closeViewer(); current = name; Array.prototype.forEach.call(document.querySelectorAll('.sb-nav button'), function(b){ b.classList.toggle('active', b.getAttribute('data-tab') === name); }); Array.prototype.forEach.call(document.querySelectorAll('.panel'), function(p){ p.classList.toggle('active', p.id === 'tab-' + name); }); var h1 = $('hTitle'); if(h1) h1.textContent = TITLES[name] || name; closeDrawer(); /* 终端页不套外框：整块内容区就是终端，所以要吃掉 main 的内边距与最大宽度。 给 body 挂个状态，CSS 那边用 body.tab-term main 收掉 —— 比 :has() 稳。 */ document.body.classList.toggle('tab-term', name === 'terminal'); /* 离开终端要把齿轮面板收掉，不然它会留在别的页面上（它是绝对定位的浮层） */ if(name !== 'terminal' && typeof tMenuToggle === 'function' && !$('tMenu').hasAttribute('hidden')) tMenuToggle(false); if(!silent){ try{ history.replaceState(null, '', '#' + name); }catch(e){} } if(name === 'overview') loadOverview(); if(name === 'sensors'){ /* 打开即采集：首次进页自动体检一次，并拉起节能持续采集 */ if(!seState.loaded) loadSensors(); seDraw(); seAutoEnter(); } if(name === 'services') loadServices(); if(name === 'processes') loadProcesses(); if(name === 'files') loadFiles(state.fs.path || ''); if(name === 'terminal' && typeof termEnter === 'function') termEnter(); } /* —— 移动端抽屉 —— */ var sbMq = window.matchMedia('(max-width:720px)'); function drawerMode(){ return sbMq.matches; } function openDrawer(){ appEl.classList.add('sb-open'); document.body.classList.add('sb-lock'); var b = $('hbg'); if(b) b.setAttribute('aria-expanded', 'true'); } function closeDrawer(){ if(!appEl.classList.contains('sb-open')) return; appEl.classList.remove('sb-open'); document.body.classList.remove('sb-lock'); var b = $('hbg'); if(b) b.setAttribute('aria-expanded', 'false'); } function toggleDrawer(){ if(appEl.classList.contains('sb-open')) closeDrawer(); else openDrawer(); } $('hbg').addEventListener('click', toggleDrawer); $('sbClose').addEventListener('click', closeDrawer); $('sbOv').addEventListener('click', closeDrawer); document.addEventListener('keydown', function(e){ if(e.key === 'Escape' || e.keyCode === 27) closeDrawer(); }); /* 抽屉开着时左滑关闭（手机上比点 × 顺手） */ (function(){ var x0 = null, y0 = null; document.addEventListener('touchstart', function(e){ if(!appEl.classList.contains('sb-open') || e.touches.length !== 1) return; x0 = e.touches[0].clientX; y0 = e.touches[0].clientY; }, {passive:true}); document.addEventListener('touchend', function(e){ if(x0 === null) return; var t = e.changedTouches[0], dx = t.clientX - x0, dy = Math.abs(t.clientY - y0); x0 = null; if(dx < -55 && dy < 70) closeDrawer(); }, {passive:true}); })(); /* 视口变宽（旋转/缩放）时清掉抽屉状态，避免残留的滚动锁 */ function syncDrawer(){ if(!drawerMode()) closeDrawer(); } if(sbMq.addEventListener) sbMq.addEventListener('change', syncDrawer); else if(sbMq.addListener) sbMq.addListener(syncDrawer); syncDrawer(); /* —— 桌面端侧栏收起（移动端忽略该状态） —— */ var SBKEY = 'ta_sb_collapsed'; function setCollapsed(on){ appEl.classList.toggle('collapsed', !!on); } try{ setCollapsed(localStorage.getItem(SBKEY) === '1'); }catch(e){} $('sbToggle').addEventListener('click', function(){ var on = !appEl.classList.contains('collapsed'); setCollapsed(on); try{ localStorage.setItem(SBKEY, on ? '1' : '0'); }catch(e){} }); /* —— 菜单点击 / hash —— */ Array.prototype.forEach.call(document.querySelectorAll('.sb-nav button'), function(b){ b.addEventListener('click', function(){ switchTab(b.getAttribute('data-tab')); window.scrollTo({top:0, behavior:'smooth'}); }); }); window.addEventListener('hashchange', function(){ var h = location.hash.replace('#',''); if(TABS.indexOf(h) >= 0 && h !== current) switchTab(h, true); }); $('btnRefresh').addEventListener('click', refreshAll); function refreshAll(){ if(current === 'overview') loadOverview(); else if(current === 'sensors') loadSensorEnv(); else if(current === 'services') loadServices(); else if(current === 'processes') loadProcesses(); else if(current === 'files') loadFiles(state.fs.path || ''); }
